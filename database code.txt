DROP DATABASE IF EXISTS inventory_db;
CREATE DATABASE inventory_db;
USE inventory_db;

-- 1. Bảng Supplier
CREATE TABLE supplier (
    supplier_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    address VARCHAR(255),
    contact VARCHAR(20),
    email VARCHAR(100),
    type VARCHAR(50),
    country VARCHAR(100),
    website VARCHAR(100),
    status ENUM('active', 'inactive') DEFAULT 'active',
    description TEXT
);

-- 2. Bảng Product
CREATE TABLE product (
    product_id INT AUTO_INCREMENT PRIMARY KEY,
    category VARCHAR(100),
    price DECIMAL(10, 2) NOT NULL,
    stock INT DEFAULT 0,
    supplier_id INT,
    manufacture_date DATE,
    expiry_date DATE,
    name VARCHAR(100),
    status ENUM('available', 'out_of_stock', 'discontinued') DEFAULT 'available',
    description TEXT,
    FOREIGN KEY (supplier_id) REFERENCES supplier(supplier_id) ON DELETE SET NULL ON UPDATE CASCADE
);

-- 3. Bảng Customer
CREATE TABLE customer (
    customer_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    phone VARCHAR(20),
    address VARCHAR(255),
    email VARCHAR(100),
    date_of_birth DATE,
    gender ENUM('Male', 'Female', 'Other'),
    registration_date DATE,
    points INT DEFAULT 0,
    status ENUM('active', 'inactive') DEFAULT 'active'
);

-- 4. Bảng Transaction (Đã sử dụng item_purchased thay vì note)
CREATE TABLE transaction (
    transaction_id INT AUTO_INCREMENT PRIMARY KEY,
    customer_id INT,
    date DATETIME DEFAULT CURRENT_TIMESTAMP,
    total_amount DECIMAL(10, 2) DEFAULT 0.00,
    payment_method ENUM('cash', 'card', 'transfer', 'e-wallet') DEFAULT 'cash',
    item_purchased VARCHAR(255),
    tax DECIMAL(5, 2) DEFAULT 0.00,
    discount DECIMAL(10, 2) DEFAULT 0.00,
    status ENUM('completed', 'pending', 'refunded', 'cancelled') DEFAULT 'completed',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (customer_id) REFERENCES customer(customer_id) ON DELETE SET NULL ON UPDATE CASCADE
);

-- 5. Bảng Involve (Chi tiết giao dịch)
CREATE TABLE involve (
    product_id INT,
    transaction_id INT,
    quantity INT DEFAULT 1,
    unit_price DECIMAL(10, 2) DEFAULT 0.00,
    PRIMARY KEY (product_id, transaction_id),
    FOREIGN KEY (product_id) REFERENCES product(product_id) ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (transaction_id) REFERENCES transaction(transaction_id) ON DELETE CASCADE ON UPDATE CASCADE
);
